@Test
public void testCaesarDecipherWrapAroundAndShiftModulo() {
    DecryptionUtils utils = new DecryptionUtils("key");
    // Shift 1 wrapping 'a' to 'z' and 'A' to 'Z'
    assertEquals("zab ZAB", utils.caesarDecipher("abc ABC", 1));
    // Shift 29 is equivalent to shift 3 (29 % 26 = 3)
    assertEquals("Hello World!", utils.caesarDecipher("Khoor Zruog!", 29));
}

@Test
public void testCaesarDecipherNonAlphabetAndEmpty() {
    DecryptionUtils utils = new DecryptionUtils("key");

}
