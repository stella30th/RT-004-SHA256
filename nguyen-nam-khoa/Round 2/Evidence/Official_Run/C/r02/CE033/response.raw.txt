@Test
public void testCaesarDecipherBasic() {
    DecryptionUtils utils = new DecryptionUtils("key");
    String decrypted = utils.caesarDecipher("Khoor, Zruog!", 3);
    assertEquals("Hello, World!", decrypted);
}

@Test
public void testCaesarDecipherShiftVariations() {
    DecryptionUtils utils = new DecryptionUtils("key");
    // Shift 0 should produce the same string
    assertEquals("Sample Text 123", utils.caesarDecipher("Sample Text 123", 0));
    // Shift 26 is equivalent to shift 0
    assertEquals("Sample Text 123", utils.caesarDecipher("Sample Text 123", 26));
    // Shift 29 is equivalent to shift 3 (29 % 26 == 3)
    assertEquals("Hello", utils.caesarDecipher("Khoor", 29));
}

@Test
public void testCaesarDecipherNonAlphabetic() {
    DecryptionUtils utils = new DecryptionUtils("key");
    String symbolsOnly = "12345!@#$%^&*()_+ \t\n";
    assertEquals(symbolsOnly, utils.caesarDecipher(symbolsOnly, 7));
}
