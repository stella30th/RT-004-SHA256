        HashMap<String, Integer> note_map = new HashMap<String, Integer>();
        note_map.put("o", 4);
        note_map.put("o|", 2);
        note_map.put(".|", 1);

        List<Integer> result = new ArrayList<Integer>();

        if (music_string.trim().equals(""))
